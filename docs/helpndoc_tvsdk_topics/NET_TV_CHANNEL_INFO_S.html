NET_TV_CHANNEL_INFO_S

通用通道信息

struct
{
UINT32 dwSize;
UINT32 dwChannel;
BYTE byEnable;
BYTE byOnline;
BYTE byStreamType;
BYTE byHasRecord;
INT32 nRecordStatus;
INT32 nDevState;
INT32 nAppProto;
INT32 nTransProto;
INT32 nMfrsType;
INT32 nCtrlPort;
INT32 nReserved[3];
CHAR szChannelName[NET_TV_LEN_64];
CHAR szDevName[NET_TV_LEN_64];
CHAR szDevType[NET_TV_LEN_64];
CHAR szSerialNum[NET_TV_LEN_64];
CHAR szFirmwareVersion[NET_TV_LEN_64];
CHAR szDeviceIP[NET_TV_IPADDR_STR_MAX_LEN];
CHAR szMac[NET_TV_LEN_64];
CHAR szSubnetMask[NET_TV_IPADDR_STR_MAX_LEN];
CHAR szMfrsName[NET_TV_LEN_64];
CHAR szAppProtoName[NET_TV_LEN_64];
CHAR szOnvifDeviceUrl[NET_TV_MAX_URL_LEN];
CHAR szPreviewMainUrl[NET_TV_MAX_URL_LEN];
CHAR szPreviewSubUrl[NET_TV_MAX_URL_LEN];
CHAR szRtspMainUrl[NET_TV_MAX_URL_LEN];
CHAR szRtspSubUrl[NET_TV_MAX_URL_LEN];
BYTE byRes[512];
};

Parameters

[in] dwSize

结构体大小

[in] dwChannel

通道号

[in] byEnable

是否启用

[in] byOnline

是否在线

[in] byStreamType

默认码流：0主码流 1子码流

[in] byHasRecord

是否有录像

[in] nRecordStatus

当前录制状态 NET_TV_RECORD_STATUS_E

[in] nDevState

设备状态

[in] nAppProto

接入协议

[in] nTransProto

传输协议

[in] nMfrsType

厂商类型

[in] nCtrlPort

管理端口

[in] nReserved

保留字段，当前转换或业务实现未使用

[in] szChannelName

通道名称

[in] szDevName

设备名称

[in] szDevType

设备类型

[in] szSerialNum

序列号

[in] szFirmwareVersion

固件版本

[in] szDeviceIP

IP地址

[in] szMac

MAC地址

[in] szSubnetMask

子网掩码

[in] szMfrsName

厂商名称

[in] szAppProtoName

协议名称

[in] szOnvifDeviceUrl

ONVIF设备服务地址

[in] szPreviewMainUrl

主码流预览地址

[in] szPreviewSubUrl

子码流预览地址

[in] szRtspMainUrl

直连主码流地址

[in] szRtspSubUrl

直连子码流地址

[in] byRes

保留字段，当前转换或业务实现未使用
