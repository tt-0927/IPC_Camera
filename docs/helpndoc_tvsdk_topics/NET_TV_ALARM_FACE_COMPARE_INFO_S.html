NET_TV_ALARM_FACE_COMPARE_INFO_S

人脸比对结果告警

struct
{
UINT32 dwAlarmType;
UINT32 dwChannel;
INT64 llTimestampMs;
INT32 nEventId;
INT32 nCompResult;
INT32 nSimilarity;
INT32 nFaceId;
CHAR szFaceLibName[NET_TV_FACE_DB_NAME_LEN];
CHAR szFaceName[NET_TV_FACE_MEMBER_NAME_LEN];
CHAR szLibFacePath[NET_TV_LEN_260];
CHAR szCapFacePath[NET_TV_LEN_260];
CHAR szCapImagePath[NET_TV_LEN_260];
BYTE byLibFaceImg[NET_TV_FACE_IMAGE_MAX_LEN];
UINT32 dwLibFaceImgLen;
BYTE byCapFaceImg[NET_TV_FACE_IMAGE_MAX_LEN];

BYTE byRes[256];
};

[in] dwChannel

通道号

[in] llTimestampMs

[in] nEventId

事件ID

[in] nCompResult

比对结果：0-不匹配 1-匹配

[in] nSimilarity

[in] nFaceId

[in] szFaceLibName

目标库名称

[in] szFaceName

人脸名称

目标库人脸图片路径

[in] szCapFacePath

抓拍人脸图片路径

[in] szCapImagePath

抓拍原图路径

[in] byLibFaceImg

目标库人脸 JPEG 二进制图片

[in] dwLibFaceImgLen

目标库人脸 JPEG 图片长度

[in] byCapFaceImg

抓拍人脸 JPEG 二进制图片

[in] dwCapFaceImgLen

抓拍人脸 JPEG 图片长度

[in] byRes

保留字段，当前转换或业务实现未使用
