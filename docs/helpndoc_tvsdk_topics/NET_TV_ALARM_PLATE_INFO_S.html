NET_TV_ALARM_PLATE_INFO_S

交通/车辆相关 (0x4000 - 0x40FF)

struct
{
UINT32 dwAlarmType;
UINT32 dwChannel;
CHAR szPlateNumber[NET_TV_LEN_32];
UINT32 dwPlateColor;
UINT32 dwVehicleType;
float fConfidence;
UINT32 dwSpeed;
UINT32 dwLaneNo;
BYTE byPlateImg[NET_TV_VEH_PLATE_IMAGE_LEN];
UINT32 dwPlateImgLen;
BYTE byRes[64];
};

[in] dwSpeed

[in] dwLaneNo

[in] byPlateImg

车牌图片

[in] dwPlateImgLen

车牌图片长度

[in] byRes

保留字段，当前转换或业务实现未使用
