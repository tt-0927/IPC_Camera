NET_TV_WIFI_STA_CONNECT_S

WIFI STA连接参数

struct
{
CHAR szSsid[NET_TV_NAME_MAX_LEN];
INT32 nSecurityMode;
CHAR szIpAddress[NET_TV_IPADDR_STR_MAX_LEN];
CHAR szPassword[NET_TV_LEN_132];
CHAR szPairwise[NET_TV_LEN_32];
INT32 nWepKeyLen;
BOOL bWepIsHex;
CHAR szAuthAlg[NET_TV_LEN_32];
INT32 nWepKeyCount;
NET_TV_WIFI_WEP_KEY_S astWepKeys[4];
CHAR szEapIdentity[NET_TV_LEN_132];
CHAR szEapPassword[NET_TV_LEN_132];
CHAR szPeapVersion[NET_TV_LEN_16];
CHAR szPhase2[NET_TV_LEN_64];
CHAR szAnonymousIdentity[NET_TV_LEN_132];
CHAR szCaCertPath[NET_TV_LEN_260];
CHAR szPeapLabel[NET_TV_LEN_32];
CHAR szTlsIdentity[NET_TV_LEN_132];
CHAR szPrivateKeyPasswd[NET_TV_LEN_132];
CHAR szEapolVersion[NET_TV_LEN_16];
CHAR szClientCertPath[NET_TV_LEN_260];
CHAR szPrivateKeyPath[NET_TV_LEN_260];
CHAR szCtrlInterface[NET_TV_LEN_260];
CHAR szInterfaceName[NET_TV_LEN_64];
BYTE byRes[256];
};

Parameters

[in] szSsid

Wi-Fi网络SSID

[in] nSecurityMode

Wi-Fi安全模式，对应NET_TV_WIFI_SECURITY_MODE_E

[in] szIpAddress

连接后分配或指定的IP地址

[in] szPassword

Wi-Fi接入密码

[in] szPairwise

成对加密方式，例如TKIP或CCMP

[in] nWepKeyLen

WEP密钥长度，取值64或128

[in] bWepIsHex

WEP密钥编码格式，true表示十六进制，false表示ASCII

[in] szAuthAlg

WEP认证算法，取值OPEN或SHARED

[in] nWepKeyCount

WEP密钥数量，最多4个

[in] astWepKeys

WEP密钥配置数组

[in] szEapIdentity

EAP认证身份

[in] szEapPassword

EAP认证密码

[in] szPeapVersion

PEAP版本

[in] szPhase2

EAP二阶段认证方式

[in] szAnonymousIdentity

EAP匿名身份

[in] szCaCertPath

CA证书路径

[in] szPeapLabel

PEAP标签

[in] szTlsIdentity

TLS认证身份

[in] szPrivateKeyPasswd

私钥密码

[in] szEapolVersion

EAPOL版本

[in] szClientCertPath

客户端证书路径

[in] szPrivateKeyPath

私钥文件路径

[in] szCtrlInterface

wpa_supplicant控制接口路径

[in] szInterfaceName

无线网卡接口名称

[in] byRes

保留字段，当前转换或业务实现未使用
