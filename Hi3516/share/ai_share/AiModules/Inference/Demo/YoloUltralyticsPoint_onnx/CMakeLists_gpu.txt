# 版本限定
cmake_minimum_required(VERSION 3.12)

# 设置程序名称
project("YoloUltralyticsPoint_onnxDemo")

include(${CMAKE_CURRENT_LIST_DIR}/../../Onnx/KeyPoint/YoloUltralyticsPoint/YoloUltralyticsPoint.cmake)

# 文件  
set (SRC_PATH
    # 后处理依赖
    ${CMAKE_CURRENT_LIST_DIR}/../../Common/json
)
foreach(item ${SRC_PATH})
    target_include_directories(${INFERENCE_NAME} INTERFACE ${item})
    aux_source_directory(${item} MODULE_SOURCES)
endforeach()

# 设置生成文件存放路径
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}/bin)

# 设置 ONNXRUNTIME_ROOT_DIR 路径
set(ONNXRUNTIME_ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../onnxruntime-linux-x64-gpu-1.16.2")

# 设置OpenCV的安装路径
# 查找 OpenCV 库
find_package(OpenCV REQUIRED)
if(NOT OpenCV_FOUND)
    message(FATAL_ERROR "OpenCV not found!")
endif()
set(ONNX_RUNTIME_LIB ${ONNXRUNTIME_ROOT_DIR}/lib/libonnxruntime.so)
# 指定 ONNX Runtime 头文件路径
include_directories(${ONNXRUNTIME_ROOT_DIR}/include)

# 编译
add_executable(${PROJECT_NAME} ${MODULE_SOURCES} main.cpp)

# 链接库路径
target_link_libraries(${PROJECT_NAME}
    YoloUltralyticsPoint_Onnx
    ${OpenCV_LIBS} 
    ${ONNX_RUNTIME_LIB}
)
