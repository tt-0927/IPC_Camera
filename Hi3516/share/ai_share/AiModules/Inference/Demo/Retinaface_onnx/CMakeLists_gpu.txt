# 版本限定
cmake_minimum_required(VERSION 3.12)

# 设置程序名称
project("Retinaface_onnxDemo")

include(${CMAKE_CURRENT_LIST_DIR}/../../Onnx/KeyPoint/Retinaface/Retinaface.cmake)

# 设置生成文件存放路径
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}/bin)

# 设置 ONNXRUNTIME_ROOT_DIR 路径
set(ONNXRUNTIME_ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../onnxruntime-linux-x64-gpu-1.16.2")

# 设置OpenCV的安装路径
# 查找 OpenCV 库
find_package(OpenCV REQUIRED)
if(NOT OpenCV_FOUND)
    message(FATAL_ERROR "OpenCV not found!")
endif()
set(ONNX_RUNTIME_LIB ${ONNXRUNTIME_ROOT_DIR}/lib/libonnxruntime.so)
# 指定 ONNX Runtime 头文件路径
include_directories(${ONNXRUNTIME_ROOT_DIR}/include)

# 编译
add_executable(${PROJECT_NAME} main.cpp)

# 链接库路径
target_link_libraries(${PROJECT_NAME}
    Retinaface_Onnx
    ${OpenCV_LIBS} 
    ${ONNX_RUNTIME_LIB}
)
