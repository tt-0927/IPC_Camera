#### Description

- 通过关闭smp/a7_linux/source/out/include/autoconf.h中相关宏定义，可裁剪security_subsys对外接口，从而减小ko和静态库大小。

/* hash 相关接口*/
CONFIG_HASH_SUPPORT
    CONFIG_HASH_PBKDF2_SUPPORT
    CONFIG_HASH_CLONE_SUPPORT

/* SYMC 相关接口*/
CONFIG_SYMC_SUPPORT
    CONFIG_SYMC_CCM_SUPPORT
    CONFIG_SYMC_GCM_SUPPORT
    CONFIG_SYMC_CMAC_SUPPORT
    CONFIG_SYMC_CBC_MAC_SUPPORT

/* PKE ECC 相关接口*/
CONFIG_PKE_ECC_SUPPORT
    CONFIG_PKE_ECC_GEN_KEY_SUPPORT
    CONFIG_PKE_ECC_ECDH_SUPPORT
    CONFIG_PKE_ECC_ECDSA_SIGN_SUPPORT
    CONFIG_PKE_ECC_ECDSA_VERIFY_SUPPORT
    CONFIG_PKE_ECC_CHECK_DOT_ON_CURVE_SUPPORT
    CONFIG_PKE_ECC_SM2_DSA_HASH_SUPPORT
    CONFIG_PKE_ECC_SM2_PUB_ENC_SUPPORT
    CONFIG_PKE_ECC_SM2_PRIV_DEC_SUPPORT

/* PKE RSA 相关接口*/
CONFIG_PKE_RSA_SUPPORT
    CONFIG_PKE_RSA_SIGN_SUPPORT
    CONFIG_PKE_RSA_VERIFY_SUPPORT
    CONFIG_PKE_RSA_PUB_ENC_SUPPORT
    CONFIG_PKE_RSA_PRIV_DEC_SUPPORT

/* 大数计算 相关接口*/
CONFIG_PKE_MOD_SUPPORT
CONFIG_PKE_EXP_MOD_SUPPORT

/* TRNG 相关接口 */
CONFIG_TRNG_GET_RANDOM_SUPPORT
CONFIG_TRNG_GET_MULTI_RANDOM_SUPPORT

/* KM 相关接口 */
CONFIG_OT_KM_SUPPORT
    CONFIG_KM_CLEAR_KEY_SUPPORT
    CONFIG_KM_EFFECTIVE_KEY_SUPPORT

注意：CONFIG_HASH_SUPPORT、CONFIG_SYMC_SUPPORT、CONFIG_PKE_ECC_SUPPORT和CONFIG_PKE_RSA_SUPPORT控制hash、symc、ecc、rsa
    功能的开启，要支持相关接口，需要先定义这几个宏
